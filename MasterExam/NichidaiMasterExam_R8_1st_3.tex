\begin{tcolorbox} %R8_1st_3
	$\fbox{3}$ 次の問いに答えよ。
\begin{enumerate}
	\item 定積分$\int_0^{2\pi} \sin^2{\theta}d\theta$を求めよ。
	\item 変数変換$r= \sin{u}$を用いて、定積分$\int_1^2 r^2\sqrt{1- r^2}dr$を求めなさい。
	\item 変数変換$\begin{cases}x= r\cos{\theta},\\ y= 2r\sin{\theta}\end{cases}$に対して、ヤコビアン$\frac{\partial(x, y)}{\partial(r, \theta)}$を求めなさい。
	\item $D= \{(x, y);\ 4x^2+ y^2\le 4\}$に対して、$\iint_D (x^2+ y^2)dxdy$を求めなさい。
	\item 変数変換$\begin{cases}x= r\sin{\theta}\cos{\phi},\\ y= r\sin{\theta}\sin{\phi},\\ z= r\cos{\theta} \end{cases}$に対して、ヤコビアン$\frac{\partial(x, y, z)}{\partial(r, \theta, \phi)}$を求めなさい。
	\item $E= \{(x, y, z);\ x^2+ y^2+ z^2\le 1\}$に対して、$\iiint_E 2\sqrt{1- x^2- y^2- z^2}dxdydz$を求めなさい。
\end{enumerate}
\end{tcolorbox}

\begin{souanswer} %R8_1st_3_ans
\begin{enumerate}
    \item 半角の公式より$\sin{2\theta}= \frac{1- \cos{2\theta}}{2}
\begin{align*}
    \int_0^{2\pi} \sin^2{x}dx
    &= \itn_0^{2\pi} \frac{1- cos{2x}}{2}dx\\
    &= \left[\frac{x}{2}- \frac{\sin{x}}{2}\right]_0^{2\pi}\\
    &= \pi
\end{align*}

    \item $r= \sin{u}$とすると。$dr= \cos{u}du$。積分区間は\beign{tabular}{l|l}$r$& $0\to 1$\\ $u$& $0\to \pi/2$\end{tabular}
\begin{algin*}
    \int_0^1r^\sqrt{1- r^2}dr
    &= int_0^{\pi/2} \sin^2{u}\sqrt{1- \sin^2{u}}\cdots\cos{u}du\\
    &= \int_0^{\pi/2} \sin^2{u}\cos^2{u}du\\
    &= \frac{1}{4}int_0^{\pi/2}\sin^2{u}du\\
    &= \frac{1}{4}\int_0^{\pi/2}\frac{1- \cos{4u}}{2}du\\
    &= \frac{1}{8}\left[u- \frac{\sin{4u}{4}}\right_0^{\pi/2}\\
    &= \frac{\pi}{16}
\end{align*}
    \item 与えられた変換に対してヤコビアンを求める。
\begin{align*}
    \frac{\partial(x, y)}{\partial(r, \theta)}
    &= \begin{vmatrix}
        \partial x/\partial r& \partial x/\partial \theta\\ \partial y/\partial r& \partial y/\partial \theta
        \end{vmatrix}\\
    &= \begin{vmatrix}
        \cos{\theta}& -r\sin{\theta}\\
        2\sin{\theta}& 2r\cos{\theta}
        \end{vmatrix}\\
    &= 2r\cos^2{\theta}+ 2r\sin^2{\theta}= 2r
\end{align*}    
    \item 積分領域$D= \{(x, y);\ 4x^2+ y^2\le 4\}$は$x^2+ \left(\frac{y}{2}\right)^2\le 1$と変換できる。極座標変換すると領域$D$は$D'= \{(r, \theta);\ 0\le r\le 1,\ 0\le \theta\le 2\pi\}$と書き換えられる。
\begin{align*}
    \iint_D(x^2+ y^2)dxdy
    &= \iint_{D'}((r\cos{\theta})^2+ (2r\sin{\theta})^2)\cdot 2rdrd\theta\\
    &= \iint_{D'}r^2(\cos^2{\theta}+ 4\sin^2{\theta})\cdot 2rdrd\theta\\
    &= \int_{\theta= 0}^{2\pi}\int_{r= 0}^1 2r^3(1+ 3\sin^2{\theta})drd\theta\\
    &= \left[\frac{1}{2}r^4\right]_{r= 0}^1\cdot 5\pi= \frac{5\pi}{2}
\end{align*}
    \item 与えられた変数変換についてヤコビアンを計算する。\label{item:R8_1st_3(5)}
\begin{align*}
    \frac{\partial (x, y, z)}{\partial (r, \theta, \phi)}
    &= \begin{vmatrix}
        \partial x/ \partial r& \partial x/ \partial \theta& \partial x/\partial \phi\\
        \partial y/ \partial r& \partial y/ \partial \theta& \partial y/\partial \phi\\
        \partial z/ \partial r& \partial z/ \partial \theta& \partial z/\partial \phi
    \end{vmatrix}\\
    &= \begin{vmatrix}
        \sin{\theta}\cos{\phi}& r\cos{\theta}\cos{\phi}& -r\sin{\theta}\sin{\phi}\\
        \sin{\theta}\sin{\phi}& r\cos{\theta}\sin{\phi}& r\sin{\theta}\cos{\phi}\\
        \cos{\theta}& -r\sin{\theta}& 0
    \end{vmatrix}
    \mathrm{第3行について余因子展開すると}
    &= \cos{\theta}\begin{vmatrix}r\cos{\theta}\cos{\phi}& -r\sin{\theta}\sin{\phi}\\ r\cos{\theta}\sin{\phi}& r\sin{\theta}\cos{\phi}\end{vmatrix}
    - (-r\sin{\theta})\begin{vmatrix}\sin{\theta}\cos{\phi}& -r\sin{\theta}\sin{\phi}\\ \sin{\theta}\sin{\phi}& r\sin{\theta}\cos{\phi}\end{vmatrix}\\
    &= \cos{\theta}(r^2\sin{\theta}\cos{\theta})+ r\sin{\theta}(r\sin^2{\theta})= r^2\sin{\theta}
\end{align*}
    \item 積分領域$E= \{(x, y, z);\ x^2+ y^2+ z^2\le 1\}$に対して、\ref{item:R8_1st_3(5)}のように極座標変換を行う。極座標変換をした積分領域$E'$は$E'= \{(r, \theta, \phi);\ 0\le r\le 1, 0\le \theta\le \pi, 0\le \phi\le 2\pi\}$となる。$dxdydz= r^2\sin{\theta}drd\theta d\phi$、被積分関数は$2\sqrt{1- (x^2+ y^2+ z^2)}= 2\sqrt{1- r^2}$。
\begin{align*}
	\iiint_E 2\sqrt{1- (x^2+ y^2+ z^2)}dxdydz
	&= \iiint_{E'} 2\sqrt{1- r^2}\cdot r^2\sin{\theta}drd\theta d\phi\\
	&= \int_{r= 0}^1 2r^2\sqrt{1- r^2}dr\int_{\theta= 0}^\pi \sin{\theta}d\theta\int_{\phi= 0}^{2\pi}d\phi\\
	&= \frrac{\pi^2}{2} 
\end{align*}    
\end{enumerate}
\end{souanswer}
