\begin{tcolorbox} %%R7_1st_3
	$\fbox{3}$ 以下の問いに答えよ．
	$g_1(x, y)= (e^y+ e^{-y})\cos{x}$, $g_2(x, y)= -(e^y- e^{-y})\sin{x}$、および$f(x, y)= \exp{(g_1(x, y))}\cos{(g_2(x, y))}$と定義する。
\begin{enumerate}
	\item $\frac{\partial f}{\partial x}$および$\frac{\partial f}{\partial y}$を求めよ。
	\item $\frac{\partial^2 f}{\partial x^2}+ \frac{\partial^2 f}{\partial y^2}$を求めよ。
	\item 次を満たす$x, y$の実数値関数$h(x, y)$が存在する場合にはその関数を求めよ。また存在しない場合にはその証明を書け。
\begin{equation*}
	\frac{\partial f}{\partial x}(x, y)= \frac{\partial h}{\partial y}(x, y) および\frac{\partial f}{\partial y}(x, y)= -\frac{\partial h}{\partial x}(x, y)
\end{equation*}
\end{enumerate}
\end{tcolorbox}

\begin{souanswer} %%R7_1st_3_ans
	$\fbox{3}$
\begin{enumerate}
	\item \label{step:R7_1st_3(1)}まず$g_1, g_2$の偏導関数を求める。
\begin{align*}
	\frac{\partial g_1}{\partial x}&= -(e^y+ e^{-y})\sin{x}\\
	\frac{\partial g_1}{\partial y}&= (e^y- e^{-y})\cos{x}\\
	\frac{\partial g_2}{\partail x}&= -(e^y+ e^{-y})\cos{x}\\
	\frac{\partail g_2}{\partail y}&= -(e^y- e^{-y})\sin{x}
\end{align*}
	合成関数の微分法より$f(x, y)= \exp{g_1}\exp{g_2}$の偏導関数は以下のように計算される。
\begin{align*}
	\frac{\partial f}{\partial x}&= -e^{(e^y+ e^{-y})\cos{x}}((e^y+ e^{-y})\sin{x}\cos{g_2(x, y)+ (e^y- e^{-y})\cos{x}\sin{g_2(x, y)})\\
	\frac{\partial f}{\partail y}&= e^{(e^y+ e^{-y}\cos{x}}((e^y- e^{-y}\cos{x}\cos{g_2(x, y)}+ (e^y+ e^{-y})\sin{x}\sin{g_2(x, y)})
\end{align*}
	\item \ref{step:R7_1st_3(1)}の結果から以下の関係式が成り立つ。
\begin{align*}
	\frac{\partial g_1}{\partial x}&= -(e^y+ e^{-y})\sin{x}= \frac{\partial g_2}{\partial y}\\
	\frac{\partial g_1}{\partial y}&= (e^y+ e^{-y})\cos{x}= -\frac{\partial g_2}{\partial x}
\end{align*}
	これはCauchy-Riemannの関係式をみたす。したがって$g_1, g_2$は調和関数。\\
	関数$f(x, y)= e^{g_1}\cos{g_2}$を$x, y$でそれぞれ偏微分して足し合わせると、合成関数の微分法より以下が得られる。
\begin{equation*}
	\frac{\partial^2 f}{\partial x^2}+ \frac{\partial^2 f}{\partial y^2}= e^{g_1}\cos{g_2}
\end{equation*}
	\item 
\end{enumerate}
\end{souanswer}
