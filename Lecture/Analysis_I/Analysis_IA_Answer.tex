\begin{tcolorbox}$\fbox{2}$
    $X$が有限加法族，$f$が$X$上の有限加法的測度ならば，$E_1, E_2, \ldots, E_n\in X$に対して以下が成り立つ．
\begin{enumerate}
    \item $E_1\cap E_2= (E_1^c\cup E_2^c)^c\in X$
    \item $\bigcup_{k= 1}^n E_k\in X$
    \item $f\left(\sum_{k= 1}^n E_k\right)= \sum_{k= 1}^n f(E_k)$
    \item $f\left(\bigcup_{k= 1}^n E_k\right)\le \sum_{k= 1}^n f(E_k)$
    \item $E_1\subset E_2$ならば$f(E_1)\le f(E_2)$
\end{enumerate}
\end{tcolorbox}

\begin{souanswer}
\begin{enumerate}
    \item de・Morgenの法則より${(E_1\cap E_2)}^c= {E_1}^c\cup {E_2}^c$．
    $E_1\cap E_2\stackrel{条件(2), (3)}{=} ({E_1}^c\cup {E_2}^c)^c\in X$

    \item 帰納法で示す．$n= 2$のとき条件(3)より成立．$n$のとき$E_1\cup \cdots\cup E_n\in X$と仮定する．$n+ 1$のとき条件(3)より$E_1\cup \cdots\cup E_n\cup E_{n+1}\in X$
    
    \item 帰納法で示す．$n= 2$のとき
\begin{equation*}
    (左辺)= f(E_1+ E_2)\stackrel{性質(5)}{=} f(E_1)+ f(E_2)= (右辺)
\end{equation*}
    次に$n$のとき成り立つと仮定し，$n+ 1$のとき成り立つことを示す．ここで$E_n$と$E_{n+ 1}$は互いに素であり，
\begin{equation*}
    (左辺)= f\left(\sum_{k= 1}^{n+ 1}E_k\right)= f\left(\sum_{k= 1}^n E_k+ E_{n+ 1}\right)= f\left(\sum_{k= 1}^n E_k\right)+ f(E_{n+ 1})= \sum_{k= 1}^{n+ 1} f(E_k)= (右辺)
\end{equation*}

    \item 与えられた集合族$E_1, E_2, \ldots, E_n\in X$に対して互いに素な集合族$F_1, F_2, \ldots, F_n\in X$を以下のように定める．
\begin{equation}
    F_1= E_1,\ F_2= E_2\setminus E_1,\ F_3= E_3\setminus (E_1\cap E_2)
\end{equation}
    $X$は有限加法族であるから，すべての$k$に対して$F_k\in X$．$F_k$の和集合と$E_k$の
    和集合は一致することと$f$の有限加法性より
\begin{equation}
    f\left(\bigcup_{k= 1}^n E_k\right)= f\left(\bigcup_{k= 1}^n F_k\right)= \sum_{k= 1}^n f(F_k)
\end{equation}
    また測度の単調性と$F_k\subset E_k$の関係から
\begin{equation}
    f(F_k)\subset f(E_k)
\end{equation}
    が成り立つ．上の式に適用すると
\begin{equation}
    \sum_{k= 1}^n f(F_k)\le \sum_{k= 1}^n f(E_k)
\end{equation}
    よって
\begin{equation}
    f\left(\bigcup_{k= 1}^n E_k\right)\le \sum_{k= 1}^n f(E_k)
\end{equation}

\item $E_1\subset E_2$であるから$E_2= E_1+ (E_2\setminus E_1)$とかける．すなわち$f(E_1)= f(E_1+ (E_2\setminus E_1))= f(E_1)+ f(E_2\setminus E_1)$．ここで測度の非負性より$f(E_2\setminus E_1)\ge 0$．よって$E_1\subset E_2$ならば$f(E_1)\le f(E_2)$
\end{enumerate}
\end{souanswer}

\begin{tcolorbox}$\fbox{4}$
    $\epsilon> 0, c_k\ge 0\ (1\le k\le N- 1)$に対し，$\delta= \frac{\epsilon}{\epsilon+ a+ c_1+ c_2+ \cdots+ c_{N- 1}}$とおく．このとき
\begin{equation*}
    0< \delta< 1,\ \delta^N+ c_1\delta^{N- 1}+ c_2\delta^{N- 2}+ \cdots+ c_{N- 1}\delta< (1+ c_1+ c_2+ \cdots+ c_{N- 1})\delta< \epsilon
\end{equation*}
    が成り立つことを示しなさい．
\end{tcolorbox}

\begin{souanswer}
    $(第一項)< (第二項)$を示す．
\begin{equation*}
    (第一項)= \delta(\delta^{N- 1}+ c_1\delta^{N- 2}+ c_2\delta^{N- 3}+ \cdots+ c_{N- 1})
\end{equation*}
    ここで$\delta$は$0< \delta< 1$であるから，$\delta^2, \delta^3, \ldots, \delta^{N- 1}< \delta$であり，
\begin{equation*}
    \delta(\delta^{N- 1}+ c_1\delta^{N- 2}+ c_2\delta^{N- 3}+ \cdots+ c_{N- 1})< (1+ c_1+ \cdots+ c_{N- 1})
\end{equation*}
    $(第二項)< (第三項)$を示す．与えられた$\delta$の定義より$(\epsilon+ 1+ c_1+ c_2+ \cdots+ c_{N- 1})\delta< \epsilon$．ここで$\epsilon> 0$であるから
\begin{equation*}
    (1+ c_1+ \cdots+ c_{N- 1})< (1+ c_1+ c_2+ \cdots+ c_{N- 1})
\end{equation*}
\end{souanswer}

\begin{tcolorbox}$\fbox{5}$
    以下の等式を示せ．
\begin{enumerate}
    \item 
\begin{enumerate_alpha}
    \item $[([A+ (-x)]\cap E)+ x]= A\cap [E+ x]$
    \item $[-([-A]\cap E)]= A\cap [-E]$
    \item $[([A+ (-x)]\cap E^c)+ x]= A\cap [E+ x]^c$
    \item $[-([-A]\cap E^c)]= A\cap [-E]^c$
\end{enumerate_alpha}
    \item 
\begin{enumerate_alpha}
    \item $\Gamma(A)= \Gamma(A\cap [E+ x])+ \Gamma(A\cap [E+ x]^c)$
    \item $\Gamma(A)= \Gamma(A\cap [-E])+ \Gamma(A\cap [-E]^c)$
\end{enumerate_alpha}
    \item $E\in \mathcal{M}_N$に対し，$\mu([E+ x])= \mu([-E])= \mu(E)$
\end{enumerate}
\end{tcolorbox}

\begin{souanswer}
\begin{enumerate}
    \item 
\begin{enumerate_alpha}
    \item \label{step:Q5_1_a}
\begin{align}
    y\in [([A+ (-x)]\cap E)+ x]
    &\iff y- x\in [A+ (-x)]\cap E\\
    &\iff (y- x\in (A+ (-x)))\land (y- x\in E)\\
    &\iff ((y- x- (-x))\in A)\land (y- x\in E)\\
    &\iff (y\in A)\land (y\in E+ x)\\
    &\iff y\in A\cap [E+ x]
\end{align}
    \item \label{step:Q5_1_b}
\begin{align*}
    y\in [-([-A]\cap E)]
    &\iff -y \in [-A]\cap E \\
    &\iff (-y \in -A) \land (-y \in E) \\
    &\iff (-(-y) \in A) \land (y \in -E) \\
    &\iff (y \in A) \land (y \in -E) \\
    &\iff y \in A\cap [-E]
\end{align*}
    \item \label{step:Q5_1_c}
\begin{align}
    y\in [([A+ (-x)]\cap E^c)+ x]
    &\iff y- x\in [A+ (-x)]\cap E^c\\
    &\iff (y- x\in (A+ (-x)))\land (y- x\in E^c)\\
    &\iff ((y- x- (-x))\in A)\land (y- x\in E^c)\\
    &\iff (y\in A)\land \neg(y\in E+ x)\\
    &\iff y\in A\cap [E+ x]^c
\end{align}
    \item \label{step:Q5_1_d}
\begin{align}
    y\in [-([-A]\cap E^c)]
    &\iff -y \in [-A]\cap E^c \\
    &\iff (-y \in -A) \land \neg(-y \in E) \\
    &\iff (-(-y) \in A) \land \neg(y \in -E) \\
    &\iff (y \in A) \land (y \in [-E]^c) \\
    &\iff y \in A\cap [-E]^c
\end{align}
\end{enumerate_alpha}
    \item 
\begin{enumerate_alpha}
    \item 
\begin{align}
    \Gamma(A)
    &= \Gamma([A+ (-x)]\cap E)+ \Gamma([A+ (-x)]\cap E^c)\\
    &= \Gamma([([A+ (-x)]\cap E)+ x])+ \Gamma([([A+ (-x)]\cap E^c)+ x])\\
    &\stackrel{\ref{step:Q5_1_a}, \ref{step:Q5_1_c}}{=} \Gamma(A\cap [E+ x])+ \Gamma(A\cap [E+ x]^c)
\end{align}
    \item 
\begin{align}
    \Gamma(A)
    &= \Gamma([-A])\\
    &= \Gamma([-A]\cap E)+ \Gamma([-A]\cap E^c)\\
    &\stackrel{\ref{step:Q5_1_b}, \ref{step:Q5_1_d}}= \Gamma(A\cap [-E])+ \Gamma(A\cap [-E]^c)
\end{align}
\end{enumerate_alpha}
    \item $\mu$は$\Gamma$の制限なので，
\begin{align}
    \mu([E+ x])&= \Gamma([E+ x])\\
    \mu([-E])&= \Gamma([-E])\\
    \mu(E)&= \Gamma(E)
\end{align}
    であることと$\Gamma([E+ x])= \Gamma([-E])= \Gamma(E)$から$\mu([E+ x])= \mu([-E])= \mu(E)$
\end{enumerate}
\end{souanswer}

\begin{tcolorbox}$\fbox{6}$
\begin{enumerate}
    \item $\{A_n\}$が単調増加ならば$\bigcap_{k= n}^\infty A_k= A_n, \bigcup_{k= n}^\infty A_k= \bigcup_{k= 1}^\infty A_k\ \ (\forall n\ge 1)$が成り立つことを示せ．
    \item $\{A_n\}$が単調減少ならば$\lim_{n\to \infty} A_n= \bigcap_{n= 1}^\infty A_n$が成り立つことを示せ．
    \item $A_{2n- 1}= [0, 1],\ A_{2n}= [1, 2]\subset \R$に対し，$\liminf_{n\to \infty} A_n,\ \limsup_{n\to \infty} A_n$を求め，これらが一致しないことを示せ．
\end{enumerate}
\end{tcolorbox}

\begin{souanswer}
\begin{enumerate}
    \item 
\begin{enumerate}
    \item $\bigcap_{k= n}^\infty A_k= A_n$を示す．$\bigcap_{k= n}^\infty A_k\subset A_n$は明らか．単調増加性より任意の$k\ge n$で$A_n\subset A_k$となり，$A_n\subset \bigcap_{k= n}^\infty= A_k$
    
    \item 集合の和を$n- 1$までと$n$以降に分割する．すなわち
\begin{equation}
    \bigcup_{k= 1}^\infty A_k= \left(\bigcup_{k= 1}^{n- 1} A_k\right)\cup \left(\bigcup_{k= n}^\infty A_k\right)
\end{equation}
    $\{A_n\}$は単調増加列であるから，任意の$1\le k\le n- 1$に対して$A_k\subset A_n$が成り立つ．だから
\begin{equation}
    \bigcup_{k= 1}^{n- 1} A_k\subset A_n\subset \bigcup_{k= n}^\infty A_k
\end{equation}
    が成り立つ．よって
\begin{equation}
    \bigcup_{k= 1}^\infty A_k= \bigcup_{k= n}^\infty A_k
\end{equation}
\end{enumerate}
    \item 
\begin{align*}
    (左辺)
    &= \liminf_{n\to \infty}A_n= \limsup_{n\to \infty}A_n\\
    &= \bigcup_{n= 1}^\infty \bigcap_{k= n}^\infty A_n= \bigcap_{n= 1}^\infty \bigcup_{k= n}^\infty A_n\\
    &= \bigcap_{n= 1}^\infty A_n\\
    &= (右辺)
\end{align*}
    \item $\liminf_{n\to \infty} A_n$から考える．任意の番号$n$について$\bigcap_{k= n}^\infty A_n$を考える．$k\ge n$では偶数，奇数のどちらもが含まれ，共通部分は
\begin{equation}
    \bigcap_{k= n}^\infty A_n= [0, 1]\cap [1, 2]= \{1\}
\end{equation}
    となる．この和集合も$\{1\}$となる．
\begin{equation}
    \liminf_{n\to \infty} A_n= \bigcup_{n= 1}^\infty \{1\}= \{1\}
\end{equation}
    次に$\limsup_{n\to \infty} A_n$について考える．任意の番号$n$について$\bigcup_{k= n}^\infty A_k$を考える．$k\ge n$にも同様に偶数，奇数のどちらも含まれるため，和集合は
\begin{equation}
    \bigcup_{k= n}^\infty A_k= [0, 1]\cup [1, 2]= [0, 2]
\end{equation}
    したがってすべての$n$についての和集合は$[0, 2]$で一定であり，その共通部分も$[0, 2]$で共通，
\begin{equation}
    \limsup_{n\to \infty} A_n= \bigcap_{n= 1}^\infty [0, 2]= [0, 2]
\end{equation}
    以上より
\begin{equation}
    \{1\}= \liminf_{n\to \infty} A_n\ne \limsup_{n\to \infty} A_n= [0, 2]
\end{equation}
    となり，一致しない．
\end{enumerate}
\end{souanswer}

\begin{tcolorbox}$\fbox{7}$
\begin{enumerate}
    \item $\exists n_0\in \N \st\mu\left(\bigcup_{n= n_0}^\infty E_n\right)< \infty$ならば
\begin{equation}
    \mu\left(\limsup_{n\to \infty} E_n\right)\ge \limsup_{n\to \infty} \mu(E_n)
\end{equation}
    \item $\exists n_0\in \N \st\mu\left(\bigcup_{n= n_0}^\infty E_n\right)< \infty$かつ$\exists \lim_{n\to \infty} E_n$ならば
\begin{equation}
    \mu\left(\lim_{n\to \infty} E_n\right)= \lim_{n\to \infty} \mu(E_n)
\end{equation}
\end{enumerate}
\end{tcolorbox}

\begin{souanswer}
\begin{enumerate}
    \item $\limsup$の定義より
\begin{align}
    \mu\left(\limsup_{n\to \infty} E_n\right)
    &= \mu\left(\bigcap_{n= 1}^\infty \bigcup_{k\ge n}^\infty E_k\right) 
\end{align}
    $\bigcup_{k\ge n} E_k= F_n$とおくと$\{F_n\}$は単調減少であり，かつ$\mu(F_{n_0})< \infty$
\begin{align}
    \mu\left(\lim_{n\to \infty} F_n\right)
    &= \lim_{n\to \infty} \mu(F_n)\\
    &= \limsup_{n\to \infty} \mu(F_n)\\
    &\ge \limsup_{n\to \infty} \mu(E_n)
\end{align}
    \item 
\begin{align}
    \mu\left(\lim_{n\to \infty} E_n\right)
    &= \mu\left(\liminf_{n\to \infty} E_n\right)\\
    &\le \liminf_{n\to \infty} \mu(E_n)\\
    &= \lim_{n\to \infty} \mu(E_n)\\
    \mu\left(\lim_{n\to \infty} E_n\right)
    &= \mu\left(\limsup_{n\to \infty} E_n\right)\\
    &\ge \limsup_{n\to \infty} \mu(E_n)\\
    &= \lim_{n\to \infty} \mu(E_n)
\end{align}
    以上より$\mu\left(\lim_{n\to \infty} E_n\right)= \lim_{n\to \infty} \mu(E_n)$
\end{enumerate}
\end{souanswer}
